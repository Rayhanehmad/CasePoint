Use the uploaded file /mnt/data/2003 MLD.docx

Extract text using python-docx:

from docx import Document

doc = Document("/mnt/data/2003 MLD.docx")
text = "\n".join([para.text for para in doc.paragraphs])


Apply improved regex pattern that keeps internal references intact:

import re

pattern = r'((?:PLD|SCMR|YLR|CLC|CLD|MLD|PCrLJ|PLC|PTD)\s+\d{4}\s+(?:[A-Z]+\s+)?\d+.*?)(?=(?:PLD|SCMR|YLR|CLC|CLD|MLD|PCrLJ|PLC|PTD)\s+\d{4}\s+(?:[A-Z]+\s+)?\d+|$)'
matches = re.findall(pattern, text, re.DOTALL)


Add merge logic for small fragments (to fix false breaks):

cleaned_citations = []
buffer = ""
for block in matches:
    if len(block.strip()) < 200:
        buffer += " " + block
    else:
        if buffer:
            cleaned_citations.append(buffer + " " + block)
            buffer = ""
        else:
            cleaned_citations.append(block)


Print debugging output with clear visual markers:

print("=== DEBUGGING CITATION EXTRACTION ===")
print(f"Total citations found: {len(cleaned_citations)}\n")

for i, c in enumerate(cleaned_citations, 1):
    start_line = f"\n===== START CITATION {i} ====="
    end_line = f"===== END CITATION {i} =====\n"
    title = c.strip().split("\n")[0][:60]
    print(start_line)
    print(f"[TITLE PREVIEW] {title}")
    print(c.strip())
    print(end_line)
